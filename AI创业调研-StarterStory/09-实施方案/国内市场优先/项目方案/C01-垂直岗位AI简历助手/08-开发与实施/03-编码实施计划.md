**Architecture:** 按已采用V2架构，生产代码仅在本idea的`coding/backend`，使用本地CLI驱动SQLite元数据和不可变原件/成果目录。领域规则、人工审阅、导出和可选模型适配分别隔离；模型关闭仍能完整人工交付。首版没有客户门户，故不创建`coding/frontend`或`coding/app`；现有07原型保留为评审材料。

**Tech Stack:** Python、SQLite、python-docx、LibreOffice、pytest。模型采用已批准Qwen Plus系列候选方向，具体国内地域模型ID经能力与授权校验后锁定；本计划不安装依赖。

```python
import hashlib
import re
from pathlib import Path

def checked_child(root: Path, *parts: str) -> Path:
    if not all(re.fullmatch(r'[A-Za-z0-9_.-]+', p) and p not in ('.', '..') for p in parts):
        raise ValueError('INVALID_PATH')
    path = root.joinpath(*parts).resolve()
    if not path.is_relative_to(root.resolve()):

    return path

def blob_digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()
```

```python
import re

def validate_fact(fact: dict) -> dict:
    for key in ('start', 'end'):
        if not re.fullmatch(r'[0-9]{4}-(0[1-9]|1[0-2])', fact[key]):
            raise ValueError('INVALID_DATE')
    if fact['start'] > fact['end']:
        raise ValueError('DATE_ORDER')
    for number in fact['numbers']:
        if '%' in number['value'] and number.get('base') in (None, ''):
            raise ValueError('MISSING_BASE')
    return dict(fact)
```

```python
def validate_claim(claim: dict, facts: dict) -> dict:
    ids = claim['fact_ids']
    if not ids or any(i not in facts or not facts[i]['confirmed'] for i in ids):
        raise ValueError('UNVERIFIED_FACT')
    allowed = {n['value'] for i in ids for n in facts[i]['numbers'] if n['confirmed']}

        raise ValueError('UNVERIFIED_NUMBER')
    return dict(claim)
```

```python
import hashlib
import json
from pathlib import Path

def verify_bundle(manifest: Path) -> None:
    meta = json.loads(manifest.read_text(encoding='utf-8'))
    for item in meta['files']:
        path = (manifest.parent / item['name']).resolve()
        if not path.is_relative_to(manifest.parent.resolve()):
            raise ValueError('INVALID_PATH')
        if not path.is_file() or hashlib.sha256(path.read_bytes()).hexdigest() != item['sha256']:
            raise ValueError('DIGEST_MISMATCH')
```

```python
from collections.abc import Callable

def generate_candidate(payload: dict, *, enabled: bool, permitted: bool,
                       transport: Callable[[dict], dict] | None) -> dict | None:
    if not enabled:
        return None
    if not permitted or transport is None:
        raise ValueError('MODEL_NOT_AUTHORIZED')
    result = transport(payload)
    if not isinstance(result, dict):
        raise ValueError('INVALID_MODEL_RESPONSE')
    return result
```
