```python
def checked_child(root: Path, *parts: str) -> Path:
    if not all(re.fullmatch(r'[A-Za-z0-9_.-]+', p) and p not in ('.', '..') for p in parts):
```

```python
from decimal import Decimal, InvalidOperation
import re

def parse_amount(text: str, *, thousands: str, places: int) -> int:
    grouping = re.escape(thousands)
    pattern = rf'-?(?:[0-9]+|[0-9]{{1,3}}(?:{grouping}[0-9]{{3}})+)(?:\.[0-9]+)?'
    if not re.fullmatch(pattern, text):
        raise ValueError('INVALID_AMOUNT')
    value = Decimal(text.replace(thousands, ''))
    scaled = value * (10 ** places)
    if not value.is_finite() or scaled != scaled.to_integral_value():
        raise ValueError('INVALID_PRECISION')
    return int(scaled)
```

拒绝重复列、空订单ID、未声明币种和精度；CSV编码由配置指定，XLSX由pandas读取。订单、退款、广告各一份schema，缺少整份广告文件以None表示。normalize_rows先检查源表头集合，再映射必需字段，不对未知列名猜测。三表入库批次键为client/task/文件类别/期间/文件SHA256，同内容换名不重复累计；新内容同期间登记supersedes，不能自动累加。

```python
def calculate(orders: list[dict], refunds: list[dict], ads: int | None) -> dict:
    keyed = {(o['shop'], o['order']): o for o in orders}
    if len(keyed) != len(orders):
        raise ValueError('DUPLICATE_ORDER')
    if len({r['id'] for r in refunds}) != len(refunds):
        raise ValueError('DUPLICATE_REFUND')
    matched = [r for r in refunds if (r['shop'], r['order']) in keyed]
    return {'paid_less_refund': sum(o['paid'] for o in orders) - sum(r['amount'] for r in matched),
            'ads': ads, 'exceptions': [r['id'] for r in refunds if r not in matched]}
```

每个指标补充source行集合、rule_version、currency、period与不可计算原因；采用先按shop/order归并退款再关联的审计明细，不把多笔退款回连造成订单倍增。新增ratio(1,0) is None、两份同内容导入后calculate相同、冻结输入复算一致的测试。未匹配退款留异常，正式交付前必须有解释和处理记录，不自动计入净指标；缺成本时无净利润字段。

```python
import hashlib
import json
from pathlib import Path

def verify_bundle(manifest: Path) -> None:
    meta = json.loads(manifest.read_text(encoding='utf-8'))
    for item in meta['files']:
        path = (manifest.parent / item['name']).resolve()
        if not path.is_relative_to(manifest.parent.resolve()):
            raise ValueError('INVALID_PATH')
        if not path.is_file() or hashlib.sha256(path.read_bytes()).hexdigest() != item['sha256']:
            raise ValueError('DIGEST_MISMATCH')
```

```python
def snapshot_database(source: Path, target: Path) -> None:
```

```python
from collections.abc import Callable

def generate_candidate(payload: dict, *, enabled: bool, permitted: bool,
                       transport: Callable[[dict], dict] | None) -> dict | None:
    if not enabled:
        return None
    if not permitted or transport is None:
        raise ValueError('MODEL_NOT_AUTHORIZED')
    result = transport(payload)
    if not isinstance(result, dict):
        raise ValueError('INVALID_MODEL_RESPONSE')
    return result
```
